cmake_minimum_required(VERSION 3.20)
project(ac97_posix C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  message(FATAL_ERROR "This regression exercises the real Linux x86-64 single-step context")
endif()
add_executable(ac97_check test_main.c)
target_include_directories(ac97_check PRIVATE ../../src)
target_compile_options(ac97_check PRIVATE -ffunction-sections -fdata-sections)
target_link_options(ac97_check PRIVATE -Wl,--gc-sections)
enable_testing()
add_test(NAME ac97_reset_write COMMAND ac97_check)
set_tests_properties(ac97_reset_write PROPERTIES TIMEOUT 10)
