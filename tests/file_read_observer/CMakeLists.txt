cmake_minimum_required(VERSION 3.20)
project(file_read_observer_test C)
set(CMAKE_C_STANDARD 11)
set(KERNEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/kernel")
add_executable(file_read_observer_test test_notify.c "${KERNEL_DIR}/file_read_observer.c")
target_include_directories(file_read_observer_test PRIVATE "${KERNEL_DIR}")
enable_testing()
add_test(NAME file_read_observer_dispatch COMMAND file_read_observer_test)
option(FILE_READ_OBSERVER_IO_TEST "Test actual kernel reads with a runtime or focused POSIX link" OFF)
if(FILE_READ_OBSERVER_IO_TEST)
  if(TARGET xboxrecomp)
    add_executable(file_read_io_test test_io.c)
    target_link_libraries(file_read_io_test PRIVATE xboxrecomp)
  elseif(MSVC)
    # Use the I/O backend's own path, overlay and lock dependencies without
    # requiring GPU/SDL/title integration.
    add_executable(file_read_io_test test_io.c log_stubs.c "${KERNEL_DIR}/kernel_file.c"
      "${KERNEL_DIR}/file_read_observer.c" "${KERNEL_DIR}/kernel_overlay.c"
      "${KERNEL_DIR}/kernel_path.c" "${KERNEL_DIR}/kernel_io.c" "${KERNEL_DIR}/xbox_ggl.c")
    target_include_directories(file_read_io_test PRIVATE "${KERNEL_DIR}/..")
    target_compile_definitions(file_read_io_test PRIVATE FILE_READ_OBSERVER_FOCUSED WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(file_read_io_test PRIVATE /Gy)
    target_link_options(file_read_io_test PRIVATE /OPT:REF)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(../../src/platform platform)
    add_executable(file_read_io_test test_io.c log_stubs.c "${KERNEL_DIR}/kernel_file.c" "${KERNEL_DIR}/file_read_observer.c")
    target_link_libraries(file_read_io_test PRIVATE platform)
    target_compile_definitions(file_read_io_test PRIVATE FILE_READ_OBSERVER_FOCUSED)
    target_compile_options(file_read_io_test PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(file_read_io_test PRIVATE -Wl,--gc-sections)
  else()
    add_subdirectory(../.. runtime)
    add_executable(file_read_io_test test_io.c)
    target_link_libraries(file_read_io_test PRIVATE xboxrecomp)
  endif()
  if(WIN32)
    target_link_libraries(file_read_io_test PRIVATE dbghelp)
  endif()
  target_include_directories(file_read_io_test PRIVATE "${KERNEL_DIR}")
  add_test(NAME file_read_observer_kernel_io COMMAND file_read_io_test)
endif()
