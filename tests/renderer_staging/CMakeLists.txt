cmake_minimum_required(VERSION 3.20)
project(renderer_staging LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Renderer staging checks exercise real Linux protected guest pages")
endif()
find_package(SDL3 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(renderer_staging_check check.c ../../src/platform/guest_memory.c ../../src/platform/win32_compat.c)
target_include_directories(renderer_staging_check PRIVATE ../../src ../../src/kernel ../../src/platform)
target_compile_options(renderer_staging_check PRIVATE -ffunction-sections -fdata-sections -UNDEBUG)
target_link_options(renderer_staging_check PRIVATE -Wl,--gc-sections)
target_link_libraries(renderer_staging_check PRIVATE SDL3::SDL3 Threads::Threads m)
set_target_properties(renderer_staging_check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
enable_testing()
add_test(NAME renderer_guest_staging COMMAND renderer_staging_check)
set_tests_properties(renderer_guest_staging PROPERTIES TIMEOUT 15)
